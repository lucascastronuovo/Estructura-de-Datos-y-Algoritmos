void mostrar(int vec[], int dim){

  int i;

  printf("Vector: ");
  for (i=0; i<dim;i++){
    
    printf("%d\t", vec[i]);

    }

  return;
}


int suma_pares(int vec[], int dim){

  int i;
  int suma=0;
  
  for (i=0; i<dim;i++){
    
    if (vec[i]%2==0)
      suma+=vec[i];

    }

  return suma;
}

void desplazar_atras(int vec[], int dim){

  int pos_final=dim;
  int i;
  
  for(i=pos_final; i>0;i--)
    vec[i]=vec[i-1];

  vec[0]=0;

  return;
}

void desplazar_adelante(int vec[], int dim){

  int pos_final=dim-1;
  int i;
  
  for(i=0; i<pos_final;i++)
    vec[i]=vec[i+1];

  vec[pos_final]=0;

  return;
}

void ordenar_ascendente(int vec[], int dim){

    int i,j;
    int aux;

  for (i=0;i<dim-1;i++){

    for (j=i+1; j<dim; j++){

        if (vec[i] > vec[j]){

          aux = vec[i];
          vec[i]=vec[j];
          vec[j]=aux;
        }

      
    }
    
  }

  return;
}
