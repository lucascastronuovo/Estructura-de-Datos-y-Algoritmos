void generar_mat(int mat[][10], int n ){
  int i, j;
  for (i=0; i<n;i++){

    
      for(j=0; j<n; j++){

        printf("\nValor [%d][%d]: ", i, j);
        scanf("%d", &mat[i][j]);
          
        }
        
      }

    return;
  }
