typedef struct pila

int promedio_pilas(int s, int c){



	int prom;

	

	if (c != 0){



		prom = s / c;



		

		

	}



	else{



		printf("No se pudo calcular el promedio");

	}



	return prom;

	

}







nodop desapilar(nodop** p)

{

	nodop* aux;

  nodop datos;



	datos=**p;



	aux=*p;



	*p=(*p)->sig;

	free(aux);



  return datos;

}









nodop* crear_nodo(int valor)

{

  nodop* u;

  u = (nodop *)malloc(sizeof(nodop));

  u->valor = valor;



  u->sig = NULL;



  return u;

}
