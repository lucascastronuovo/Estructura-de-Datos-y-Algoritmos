nodop* crear_nodo(int dato )
{
   nodop* u;
   u = (nodop *)malloc(sizeof(nodop));
   u->a = dato;

   u->sig = NULL;

   return u;
}

{
    printf("apilar_nodo\n");


  	u->sig = *p;

}
