#include <stdio.h>


void mostrar_vec(float vec[15]) {

	int i;

	printf("Arreglo: ");
	
	for (i = 0; i < 15; i++){

		printf("\n %f", vec[i]);
		
	}

	
}

void cargar_vec(float vec[15]) {

	int i;

	for (i = 0; i < 15; i++) {

		printf("Valor %d: ", i);
		scanf("%f", &vec[i]);
				
	}




	
}


int main() {

	int i;

	float vec[15];

	float *p;

	cargar_vec(vec);

	mostrar_vec(vec);

	p = &vec[0];

	for (i = 0; i < 15; i++){

		printf("\n En la dirección %p se encuentra el valor %f", (&p + i), *(p + i));		
	}

	
	
		
    return 0;
}
