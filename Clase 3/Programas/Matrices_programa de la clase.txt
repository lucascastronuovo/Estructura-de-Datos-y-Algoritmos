#include <stdio.h>
#include <string.h>
void cargar(int mat[][10],int d){
	int i,j;
	for(i=0;i<d;i++){
		for(j=0;j<d;j++){
			printf("\n ingrese valor %d %d  ",i,j);
			scanf("%d",&mat[i][j]);
		}
	}

}
void mostrar(int mat[][10],int d){
	int i,j;
	for(i=0;i<d;i++){
		
		for(j=0;j<d;j++){
			printf("\t %d  ",mat[i][j]);
			
		}
		printf("\n");
	}

}
void maximo(int mat[][10],int d){
	int i,j,max=mat[0][0];
	for(i=0;i<d;i++){
		for(j=0;j<d;j++){
			if(mat[i][j]>max){
				max=mat[i][j];			}
			
	}
		printf("\n el valor maximo de la fila %d es %d",i,max);
		max=mat[i+1][0];

		}
	}





int main() {
	int mat[10][10],dim;
	printf("ingrese dimension");
	scanf("%d",&dim);
	cargar (mat,dim);
	mostrar(mat,dim);
	maximo (mat,dim);
    return 0;
}
