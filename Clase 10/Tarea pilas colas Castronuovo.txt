#include<stdio.h>
#include<stdlib.h>
#include<string.h>






typedef struct pila
{
	int valor;

    struct pila *sig;
}nodop;


int promedio_pilas(nodop** p,int c){

	int prom, sum=0, cont=0 ;
	
	while(p!=NULL){

	nodop* aux;
    nodop v;

	v=**p;

	aux=*p;

	*p=(*p)->sig;
	

	if (c <= 4){

		sum+= aux->valor;
		cont++;
		c--;
		
	}
		
	free(aux);
		
	}

	if (cont == 0){
		printf("No se pudo calcular el promedio");
		
	}
	else{

		prom = sum / cont;

		
	}
	
	return prom;
}






nodop* crear_nodo(int valor)
{
   nodop* u;
   u = (nodop *)malloc(sizeof(nodop));
   u->valor = valor;

   u->sig = NULL;

   return u;
}
