{

      *contp=*contp+1;
    }

  	printf("%d \n", aux.num);
}

nodo* crear_nodo(int dato)
{
   nodo* u;
   u = (nodo *)malloc(sizeof(nodo));

   u->num = dato;

   u->siguiente = NULL;

   return u;
}
