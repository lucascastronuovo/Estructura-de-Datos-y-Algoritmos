#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/*
Desarrollar un programa en C que permita:

Crear una lista simplemente enlazada con el número de legajo,
nombre y apellido y cantidad de materias aprobadas
de los alumnos de una facultad. La carga de datos
finaliza cuando el número de legajo es igual a cero.

Mostrar la lista.
Mostrar cuántos alumnos tiene más de 10 materias aprobadas.
Generar una nueva lista con aquellos alumnos que
tiene menos de 3 materias aprobadas.

*/

typedef struct lista_alumnos {

	int legajo;
	char nombre[20];
	char apellido[20];
	int materias_aprob;
	struct lista_alumnos *sig;

}nodon;



void crear_lista_alumno(nodon *h) {

	printf("Ingrese legajo: ");
	scanf("%d", &h->legajo);

	while(h->legajo != 0){

		printf("\n Nombre: ");
		scanf("%s",h->nombre);
		
		
		
		printf("\n Apellido: ");
		scanf("%s",h->apellido);
		

		printf("\n Materias aprobadas: ");
		scanf("%d",&h->materias_aprob);
		h->sig=(nodon *)malloc(sizeof(nodon));
		h=h->sig;
		
		printf("\n Ingrese legajo: ");
		scanf("%d", &h->legajo);
	}
	

	h->sig = NULL;
	
	
	return;	
}


void mostrar_lista_alumno(nodon *h){

	while(h->sig != NULL) {
		
		printf("Legajo: %d\n", h->legajo);		
		printf("Nombre: %s\n", h->nombre);
		printf("Apellido: %s\n", h->apellido);
		printf("Materias Aprobadas: %d\n", h->materias_aprob);
		
        printf("\n");
		

		h=h->sig;
			
		
	}

	return;
}
