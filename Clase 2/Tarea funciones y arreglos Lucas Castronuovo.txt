int i, j, aux;
	
	for (i = 0; i < dim; i++) {

		for (j= i+1; j < dim; j++){

			if (v[i] < v[j]) {

				aux = v[i];
				v[i] = v[j];
				v[j] = aux;
				
			}

			
		}


		
	}

	return;	
}
