void mostrar (charlynode *hoja, int *sum, int *min, int *raiz)
{
      
    if (hoja!=NULL)
    {
        
        
        mostrar(hoja->izq, sum, min, raiz);
        printf("%d ",hoja->dato);
        if (hoja->izq != NULL & hoja->der != NULL & hoja->dato%2==0)
        *sum=*sum+hoja->dato;

        if (hoja->dato%2==0 && hoja->dato<*min && *raiz%hoja->dato==0)
        *min=hoja->dato;


        mostrar(hoja->der, sum, min, raiz);

    }
    return;
}

return;
}
